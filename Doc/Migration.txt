	For the first time we have to define the structure of a table mainly using the forge (later for modify ) then run the migration 


	defining the schema using the forge --> 


		$fields = [
		    'id' => [
		        'type'           => 'INT',
		        'constraint'     => 5,
		        'unsigned'       => true,
		        'auto_increment' => true,
		    ],
		    'title' => [
		        'type'       => 'VARCHAR',
		        'constraint' => '100',
		        'unique'     => true,
		    ],
		    'author' => [
		        'type'       => 'VARCHAR',
		        'constraint' => 100,
		        'default'    => 'King of Town',
		    ],
		    'description' => [
		        'type' => 'TEXT',
		        'null' => true,
		    ],
		    'status' => [
		        'type'       => 'ENUM',
		        'constraint' => ['publish', 'pending', 'draft'],
		        'default'    => 'pending',
		    ],
		];
		$forge->addField($fields);
